async ({ tab }, options) => {
	// `screenshot` is the acquisition's own option; every other key is observe()'s.
	const { screenshot, ...observe } = options;
	const state = {};
	const partial = (field, error) => {
		if (tab.signal?.aborted) throw error;
		state[field] = error instanceof Error ? error.message : String(error);
	};
	try {
		// observe() prints the tree itself; the state only carries what code needs.
		state.initialObservation = await tab.observe(observe);
	} catch (error) {
		partial("inspectionError", error);
	}
	if (screenshot !== false) {
		try {
			state.initialScreenshot = await tab.screenshot();
		} catch (error) {
			partial("screenshotError", error);
		}
	}
	const errors = {};
	for (const field of ["inspectionError", "screenshotError"]) if (state[field]) errors[field] = state[field];
	if (Object.keys(errors).length > 0) display(errors);
	return state;
}
