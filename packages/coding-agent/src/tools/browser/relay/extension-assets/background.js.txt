// extension/debugger-ownership.ts
async function ownedDebuggerTabs(targets, probe) {
  const owned = await Promise.all(targets.map(async (target) => {
    if (!target.attached || target.tabId === undefined)
      return;
    try {
      await probe(target.tabId);
      return target.tabId;
    } catch {
      return;
    }
  }));
  return owned.filter((tabId) => tabId !== undefined);
}

// extension/popup-placement.ts
var NO_WINDOW = -1;
var TRANSIENT_NO_FOCUS_MS = 100;

class PopupPlacement {
  #chrome;
  #selected = new Map;
  #generation = new Map;
  #displacedTab = new Map;
  #popups = new Set;
  #focused;
  #focusedBefore = new Map;
  #awaitingFocus = new Map;
  #noneAt = Number.NEGATIVE_INFINITY;
  #beforeNone;
  #now;
  constructor(chrome2, now = Date.now) {
    this.#chrome = chrome2;
    this.#now = now;
  }
  seed(windowId, tabId, focused) {
    if (!this.#selected.has(windowId))
      this.#selected.set(windowId, tabId);
    if (focused && this.#focused === undefined)
      this.#focused = windowId;
  }
  activated(windowId, tabId) {
    const generation = (this.#generation.get(windowId) ?? 0) + 1;
    this.#generation.set(windowId, generation);
    if (this.#displacedTab.has(tabId))
      this.#popups.delete(tabId);
    this.#displacedTab.set(tabId, { previous: this.#selected.get(windowId), generation });
    this.#selected.set(windowId, tabId);
  }
  focusChanged(windowId) {
    if (windowId === NO_WINDOW) {
      this.#beforeNone = this.#focused;
      this.#noneAt = this.#now();
      this.#focused = NO_WINDOW;
      return;
    }
    const previous = this.#lastFocused();
    const back = previous === NO_WINDOW ? undefined : this.#awaitingFocus.get(windowId);
    this.#awaitingFocus.clear();
    if (back !== undefined)
      this.#chrome.focusWindow(back).catch(() => {
        return;
      });
    this.#focusedBefore.set(windowId, previous);
    this.#focused = windowId;
  }
  #lastFocused() {
    if (this.#focused === NO_WINDOW && this.#now() - this.#noneAt < TRANSIENT_NO_FOCUS_MS)
      return this.#beforeNone;
    return this.#focused;
  }
  opened(tabId) {
    this.#popups.add(tabId);
  }
  openedFrom(sourceTabId, tabId, driven) {
    if (driven || this.#popups.has(sourceTabId))
      this.opened(tabId);
  }
  removed(tabId) {
    this.#displacedTab.delete(tabId);
    this.#popups.delete(tabId);
  }
  windowRemoved(windowId) {
    this.#selected.delete(windowId);
    this.#generation.delete(windowId);
    this.#focusedBefore.delete(windowId);
    this.#awaitingFocus.delete(windowId);
    if (this.#beforeNone === windowId)
      this.#beforeNone = undefined;
  }
  async background(tabId) {
    if (!this.#popups.has(tabId))
      return;
    const tab = await this.#chrome.getTab(tabId);
    if (!tab)
      return;
    const window = await this.#chrome.getWindow(tab.windowId);
    if (window.type === "popup") {
      if (this.#focused === tab.windowId) {
        if (!window.focused)
          return;
        const before = await this.#pastPopupWindows(this.#focusedBefore.get(tab.windowId));
        if (this.#focused !== tab.windowId)
          return;
        if (before === undefined) {
          const last = await this.#chrome.lastFocusedNormalWindow();
          if (last !== undefined && last !== tab.windowId && this.#focused === tab.windowId) {
            await this.#chrome.focusWindow(last);
          }
        } else if (before !== NO_WINDOW)
          await this.#chrome.focusWindow(before);
      } else {
        const current = this.#lastFocused();
        if (current === undefined || current === NO_WINDOW || this.#focusedBefore.has(tab.windowId))
          return;
        const back = await this.#pastPopupWindows(current);
        if (this.#lastFocused() !== current || this.#focusedBefore.has(tab.windowId))
          return;
        if (back === undefined || back === NO_WINDOW)
          return;
        if (window.focused)
          await this.#chrome.focusWindow(back);
        else
          this.#awaitingFocus.set(tab.windowId, back);
      }
      return;
    }
    const own = this.#displacedTab.get(tabId);
    if (!tab.active || !own || this.#generation.get(tab.windowId) !== own.generation)
      return;
    const seen = new Set([tabId]);
    let previous = own.previous;
    while (previous !== undefined && this.#popups.has(previous) && !seen.has(previous)) {
      seen.add(previous);
      previous = this.#displacedTab.get(previous)?.previous;
    }
    if (previous === undefined || seen.has(previous))
      return;
    const target = await this.#chrome.getTab(previous);
    if (target?.windowId !== tab.windowId || this.#generation.get(tab.windowId) !== own.generation)
      return;
    await this.#chrome.selectTab(previous);
  }
  async#pastPopupWindows(windowId) {
    if (windowId === undefined || windowId === NO_WINDOW)
      return windowId;
    const popupWindows = new Set;
    for (const popup of this.#popups) {
      const tab = await this.#chrome.getTab(popup);
      if (!tab || popupWindows.has(tab.windowId))
        continue;
      const window = await this.#chrome.getWindow(tab.windowId).catch(() => {
        return;
      });
      if (window?.type === "popup")
        popupWindows.add(tab.windowId);
    }
    const seen = new Set;
    let current = windowId;
    while (current !== undefined && current !== NO_WINDOW && popupWindows.has(current)) {
      if (seen.has(current))
        return NO_WINDOW;
      seen.add(current);
      current = this.#focusedBefore.get(current);
    }
    return current;
  }
}

// extension/background.ts
var DEFAULT_PORT = 9224;
var PING_INTERVAL_MS = 20000;
var RECONNECT_MIN_MS = 1000;
var RECONNECT_MAX_MS = 1e4;
var REMOVE_TAB_ATTEMPTS = 3;
var REMOVE_TAB_RETRY_MS = 500;
var ws = null;
var reconnectDelay = RECONNECT_MIN_MS;
var pingTimer = null;
var relayInitiatedDetachTabs = new Set;
var debuggerTabs = new Set;
var placement = new PopupPlacement({
  getTab: (tabId) => chrome.tabs.get(tabId).catch(() => null),
  selectTab: (tabId) => chrome.tabs.update(tabId, { active: true }),
  getWindow: (windowId) => chrome.windows.get(windowId),
  focusWindow: (windowId) => chrome.windows.update(windowId, { focused: true }),
  lastFocusedNormalWindow: () => chrome.windows.getLastFocused({ windowTypes: ["normal"] }).then((window) => window.id).catch(() => {
    return;
  })
});
var instanceId = null;
async function ensureInstanceId() {
  if (instanceId)
    return instanceId;
  const key = "relayInstanceId";
  const stored = await chrome.storage.local.get({ [key]: "" });
  const existing = stored[key];
  instanceId = typeof existing === "string" && existing.length > 0 ? existing : crypto.randomUUID();
  await chrome.storage.local.set({ [key]: instanceId });
  return instanceId;
}
var browserSession = null;
function ensureBrowserSession() {
  browserSession ??= (async () => {
    const key = "relayBrowserSession";
    const stored = await chrome.storage.session.get({ [key]: "" });
    const existing = stored[key];
    if (typeof existing === "string" && existing.length > 0)
      return existing;
    const created = crypto.randomUUID();
    await chrome.storage.session.set({ [key]: created });
    return created;
  })();
  return browserSession;
}
async function loadSettings() {
  const stored = await chrome.storage.local.get({ port: DEFAULT_PORT, token: "" });
  const port = Number(stored.port);
  return {
    port: Number.isInteger(port) && port > 0 && port <= 65535 ? port : DEFAULT_PORT,
    token: typeof stored.token === "string" ? stored.token : ""
  };
}
function snapshot(tab) {
  if (tab.id === undefined)
    return null;
  return {
    tabId: tab.id,
    url: tab.url ?? tab.pendingUrl ?? "",
    title: tab.title ?? "",
    active: tab.active,
    discarded: tab.discarded === true,
    windowId: tab.windowId,
    pinned: tab.pinned,
    groupId: tab.groupId
  };
}
var ompGroupTitle = null;
var groupOps = Promise.resolve();
function enqueueGroupOp(fn) {
  const result = groupOps.then(fn, fn);
  groupOps = result.catch(() => {});
  return result;
}
async function groupTabs(tabIds, title, color) {
  ompGroupTitle = title;
  chrome.storage.session.set({ ompGroupTitle: title });
  const byWindow = new Map;
  for (const tabId of tabIds) {
    try {
      const tab = await chrome.tabs.get(tabId);
      if (tab.pinned || tab.id === undefined)
        continue;
      const bucket = byWindow.get(tab.windowId) ?? [];
      bucket.push(tab.id);
      byWindow.set(tab.windowId, bucket);
    } catch {}
  }
  const grouped = {};
  for (const [windowId, ids] of byWindow) {
    const existing = await chrome.tabGroups.query({ title, windowId });
    let groupId;
    if (existing[0]) {
      groupId = existing[0].id;
      for (const dupe of existing.slice(1)) {
        const dupeTabs = await chrome.tabs.query({ groupId: dupe.id });
        const dupeIds = dupeTabs.map((tab) => tab.id).filter((id) => id !== undefined);
        if (dupeIds.length > 0)
          await chrome.tabs.group({ tabIds: dupeIds, groupId });
      }
      await chrome.tabs.group({ tabIds: ids, groupId });
    } else {
      groupId = await chrome.tabs.group({ tabIds: ids });
    }
    await chrome.tabGroups.update(groupId, { title, color });
    for (const id of ids)
      grouped[String(id)] = groupId;
  }
  return { grouped };
}
async function restoreGroups() {
  if (!ompGroupTitle) {
    const stored = await chrome.storage.session.get({ ompGroupTitle: "" }).catch(() => ({ ompGroupTitle: "" }));
    ompGroupTitle = typeof stored.ompGroupTitle === "string" && stored.ompGroupTitle ? stored.ompGroupTitle : null;
  }
  if (!ompGroupTitle)
    return;
  const groups = await chrome.tabGroups.query({ title: ompGroupTitle }).catch(() => []);
  for (const group of groups) {
    const tabs = await chrome.tabs.query({ groupId: group.id }).catch(() => []);
    const ids = tabs.map((tab) => tab.id).filter((id) => id !== undefined);
    if (ids.length > 0)
      await chrome.tabs.ungroup(ids).catch(() => {});
  }
}
function post(msg) {
  if (ws?.readyState === WebSocket.OPEN)
    ws.send(JSON.stringify(msg));
}
async function removeTab(tabId) {
  for (let attempt = 1;; attempt++) {
    try {
      await chrome.tabs.remove(tabId);
      return;
    } catch (error) {
      if (/No tab with id/.test(String(error)))
        return;
      if (attempt >= REMOVE_TAB_ATTEMPTS)
        throw error;
      const pause = Promise.withResolvers();
      setTimeout(pause.resolve, REMOVE_TAB_RETRY_MS);
      await pause.promise;
    }
  }
}
async function setBadge(connected) {
  try {
    await chrome.action.setBadgeText({ text: connected ? "on" : "off" });
    await chrome.action.setBadgeBackgroundColor({ color: connected ? "#1a7f37" : "#8b8b8b" });
  } catch {}
}
async function buildHello() {
  const [tabs, targets, lastFocused] = await Promise.all([
    chrome.tabs.query({}),
    chrome.debugger.getTargets(),
    chrome.windows.getLastFocused().catch(() => null)
  ]);
  const snapshots = [];
  for (const tab of tabs) {
    const snap = snapshot(tab);
    if (snap)
      snapshots.push(snap);
    if (tab.active && tab.id !== undefined) {
      placement.seed(tab.windowId, tab.id, lastFocused?.focused === true && lastFocused.id === tab.windowId);
    }
  }
  const attachedTabIds = await ownedDebuggerTabs(targets, (tabId) => chrome.debugger.sendCommand({ tabId }, "Target.getTargetInfo"));
  for (const tabId of attachedTabIds)
    debuggerTabs.add(tabId);
  const versionMatch = /Chrome\/[\d.]+/.exec(navigator.userAgent);
  return {
    t: "hello",
    instanceId: await ensureInstanceId(),
    browserSession: await ensureBrowserSession(),
    userAgent: navigator.userAgent,
    browserVersion: versionMatch?.[0] ?? "Chrome/unknown",
    discardedTabsProtocol: 1,
    ompVersion: typeof OMP_INSTALL_VERSION === "string" ? OMP_INSTALL_VERSION : undefined,
    reloadProtocol: EXTENSION_RELOAD_PROTOCOL_VERSION,
    tabs: snapshots,
    attachedTabIds
  };
}
async function runRpc(msg, origin) {
  switch (msg.op) {
    case "attach":
      await chrome.debugger.attach({ tabId: msg.tabId }, "1.3");
      debuggerTabs.add(msg.tabId);
      return {};
    case "detach":
      relayInitiatedDetachTabs.add(msg.tabId);
      try {
        await chrome.debugger.detach({ tabId: msg.tabId });
        debuggerTabs.delete(msg.tabId);
        return {};
      } catch (error) {
        relayInitiatedDetachTabs.delete(msg.tabId);
        throw error;
      }
    case "send":
      return await chrome.debugger.sendCommand(msg.sessionId ? { tabId: msg.tabId, sessionId: msg.sessionId } : { tabId: msg.tabId }, msg.method, msg.params);
    case "createTab": {
      const tab = await chrome.tabs.create({ url: msg.url, active: msg.active ?? true });
      const snap = snapshot(tab);
      if (!snap)
        throw new Error("created tab has no id");
      if (origin.readyState !== WebSocket.OPEN)
        removeTab(snap.tabId).catch(() => {});
      return { tab: snap };
    }
    case "removeTab":
      await removeTab(msg.tabId);
      return {};
    case "activateTab": {
      const tab = await chrome.tabs.get(msg.tabId);
      await chrome.windows.update(tab.windowId, { focused: true });
      await chrome.tabs.update(msg.tabId, { active: true });
      return {};
    }
    case "backgroundTab":
      await placement.background(msg.tabId);
      return {};
    case "group":
      return await enqueueGroupOp(() => groupTabs(msg.tabIds, msg.title, msg.color));
    case "ungroup":
      await enqueueGroupOp(() => chrome.tabs.ungroup(msg.tabIds).catch(() => {}));
      return {};
  }
}
function handleRelayMessage(socket, raw) {
  let msg;
  try {
    msg = JSON.parse(raw);
  } catch {
    return;
  }
  if (msg.t === "pong")
    return;
  if (msg.t === "reload") {
    chrome.runtime.reload();
    return;
  }
  const answer = (reply) => {
    if (socket.readyState === WebSocket.OPEN)
      socket.send(JSON.stringify(reply));
  };
  runRpc(msg, socket).then((result) => answer({ t: "rpcResult", id: msg.id, ok: true, result })).catch((err) => {
    answer({ t: "rpcResult", id: msg.id, ok: false, error: err instanceof Error ? err.message : String(err) });
  });
}
function scheduleReconnect() {
  const delay = reconnectDelay;
  reconnectDelay = Math.min(reconnectDelay * 2, RECONNECT_MAX_MS);
  setTimeout(() => void connect(), delay);
}
async function connect() {
  if (ws && (ws.readyState === WebSocket.OPEN || ws.readyState === WebSocket.CONNECTING))
    return;
  const settings = await loadSettings();
  const url = `ws://127.0.0.1:${settings.port}/ext${settings.token ? `?token=${encodeURIComponent(settings.token)}` : ""}`;
  const socket = new WebSocket(url);
  ws = socket;
  socket.onopen = () => {
    reconnectDelay = RECONNECT_MIN_MS;
    setBadge(true);
    buildHello().then((hello) => post(hello));
    clearInterval(pingTimer ?? undefined);
    pingTimer = setInterval(() => post({ t: "ping" }), PING_INTERVAL_MS);
  };
  socket.onmessage = (event) => {
    if (typeof event.data === "string")
      handleRelayMessage(socket, event.data);
  };
  socket.onclose = () => {
    if (ws !== socket)
      return;
    ws = null;
    if (pingTimer !== null) {
      clearInterval(pingTimer);
      pingTimer = null;
    }
    setBadge(false);
    restoreGroups();
    scheduleReconnect();
  };
  socket.onerror = () => {
    socket.close();
  };
}
chrome.debugger.onEvent.addListener((source, method, params) => {
  if (source.tabId === undefined)
    return;
  post({ t: "cdpEvent", tabId: source.tabId, sessionId: source.sessionId, method, params });
});
chrome.debugger.onDetach.addListener((source, reason) => {
  if (source.tabId === undefined)
    return;
  const relayInitiated = relayInitiatedDetachTabs.delete(source.tabId);
  debuggerTabs.delete(source.tabId);
  post({ t: "detached", tabId: source.tabId, reason, relayInitiated });
});
chrome.tabs.onCreated.addListener((tab) => {
  const snap = snapshot(tab);
  if (snap)
    post({ t: "tabCreated", tab: snap });
});
chrome.tabs.onUpdated.addListener((_tabId, _changeInfo, tab) => {
  const snap = snapshot(tab);
  if (snap)
    post({ t: "tabUpdated", tab: snap });
});
chrome.tabs.onActivated.addListener(({ tabId, windowId }) => {
  placement.activated(windowId, tabId);
  chrome.tabs.get(tabId).then((tab) => {
    const snap = snapshot(tab);
    if (snap)
      post({ t: "tabUpdated", tab: snap });
  }).catch(() => {});
});
chrome.tabs.onRemoved.addListener((tabId) => {
  placement.removed(tabId);
  debuggerTabs.delete(tabId);
  post({ t: "tabRemoved", tabId });
});
chrome.webNavigation.onCreatedNavigationTarget.addListener(({ sourceTabId, tabId }) => {
  placement.openedFrom(sourceTabId, tabId, debuggerTabs.has(sourceTabId));
  post({ t: "popupCreated", tabId, sourceTabId });
});
chrome.windows.onFocusChanged.addListener((windowId) => placement.focusChanged(windowId));
chrome.windows.onRemoved.addListener((windowId) => placement.windowRemoved(windowId));
chrome.alarms.create("omp-relay-keepalive", { periodInMinutes: 0.5 });
chrome.alarms.onAlarm.addListener((alarm) => {
  if (alarm.name === "omp-relay-keepalive")
    connect();
});
chrome.storage.onChanged.addListener((_changes, areaName) => {
  if (areaName !== "local")
    return;
  ws?.close();
  connect();
});
chrome.action.onClicked.addListener(() => void chrome.runtime.openOptionsPage());
chrome.runtime.onInstalled.addListener(() => void connect());
chrome.runtime.onStartup.addListener(() => void connect());
connect();
